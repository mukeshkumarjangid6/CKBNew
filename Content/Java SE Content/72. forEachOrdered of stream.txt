package test;

import java.util.Arrays;
import java.util.Comparator;
import java.util.List;

public class TestExample {

	public static void main(String[] args) {
		List<Integer> list = Arrays.asList(2, 14, 4, 6, 8, 10);
		//forEachOrdered(Consumer<? super T> action): Performs an action for each element of this stream
		list.stream().forEachOrdered(System.out::println);
		System.out.println();
		//Returns a stream consisting of the elements of this stream, sorted according to the provided Comparator.
		list.stream().sorted(Comparator.reverseOrder()).forEachOrdered(System.out::println);
	}
}
