//1st Way
package test;
import java.util.stream.Stream;

public class TestExample {
	public static void main(String[] args) {
		// Creating a Stream of Strings
		Stream<String> stream = Stream.of("Java", "Android", "Kotlin", "Python");

		// Using Stream filter(Predicate predicate)
		// to get a stream consisting of the
		// elements ending with s
		stream.filter(str -> str.endsWith("n")).forEach(System.out::println);
	}
}

//2nd Way
package test;

import java.util.Arrays;
import java.util.List;

public class TestExample {
	public static void main(String[] args) {
		// Creating a list of Strings
		List<String> list = Arrays.asList("Java", "Android", "Kotlin", "Python");

		// Using Stream filter(Predicate predicate)
		// to get a stream consisting of the
		// elements ending with s
		list.stream().filter(str -> str.endsWith("n")).forEach(System.out::println);
	}
}
